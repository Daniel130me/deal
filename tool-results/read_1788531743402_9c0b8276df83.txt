     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import {
     5→  ArrowLeft,
     6→  ArrowRight,
     7→  CalendarDays,
     8→  Check,
     9→  Copy,
    10→  Loader2,
    11→  MapPin,
    12→  Minus,
    13→  PartyPopper,
    14→  Plus,
    15→  X,
    16→} from "lucide-react";
    17→import { toast } from "sonner";
    18→import { AppCanvas } from "@/components/app/chrome";
    19→import { Button } from "@/components/ui/button";
    20→import { Input } from "@/components/ui/input";
    21→import { Textarea } from "@/components/ui/textarea";
    22→import { api } from "@/lib/api";
    23→import { useApp } from "@/components/app/context";
    24→import { formatNaira, formatDate, type ClientRequest, type Deal, type Service } from "@/lib/types";
    25→import { cn } from "@/lib/utils";
    26→
    27→const WIZARD_STEPS = ["Deal details", "Scope", "Terms", "Review"];
    28→
    29→interface WizardState {
    30→  title: string;
    31→  clientName: string;
    32→  clientContact: string;
    33→  summary: string;
    34→  eventDate: string;
    35→  location: string;
    36→  message: string;
    37→  serviceTitle: string;
    38→  scope: string;
    39→  deliverables: string[];
    40→  price: number;
    41→  depositPercent: number;
    42→  startDate: string;
    43→  dueDate: string;
    44→  revisions: number;
    45→}
    46→
    47→export default function WizardScreen({ requestParam }: { requestParam: string | null }) {
    48→  const { user, navigate } = useApp();
    49→  const requestId = requestParam?.startsWith("r-") ? requestParam.slice(2) : null;
    50→
    51→  const [step, setStep] = useState(0);
    52→  const [busy, setBusy] = useState(false);
    53→  const [services, setServices] = useState<Service[]>([]);
    54→  const [linkedRequest, setLinkedRequest] = useState<ClientRequest | null>(null);
    55→  const [deliverableDraft, setDeliverableDraft] = useState("");
    56→  const [sentDeal, setSentDeal] = useState<Deal | null>(null);
    57→
    58→  const [form, setForm] = useState<WizardState>({
    59→    title: "",
    60→    clientName: "",
    61→    clientContact: "",
    62→    summary: "",
    63→    eventDate: "",
    64→    location: "",
    65→    message: "",
    66→    serviceTitle: "",
    67→    scope: "",
    68→    deliverables: [],
    69→    price: 0,
    70→    depositPercent: 50,
    71→    startDate: "",
    72→    dueDate: "",
    73→    revisions: 2,
    74→  });
    75→
    76→  const set = <K extends keyof WizardState>(key: K, value: WizardState[K]) =>
    77→    setForm((prev) => ({ ...prev, [key]: value }));
    78→
    79→  useEffect(() => {
    80→    if (!user) return;
    81→    api.deals(user.id).catch(() => undefined);
    82→    fetch(`/api/users/${user.id}/services`)
    83→      .then((r) => r.json())
    84→      .then((d) => setServices(d.services ?? []))
    85→      .catch(() => undefined);
    86→    if (requestId) {
    87→      api
    88→        .requestDetail(requestId)
    89→        .then(({ request: req, service }) => {
    90→          setLinkedRequest(req);
    91→          setForm((prev) => ({
    92→            ...prev,
    93→            title: service ? `${service.title}` : "Custom deal",
    94→            serviceTitle: service?.title ?? "",
    95→            clientName: req.clientName,
    96→            clientContact: req.clientContact,
    97→            eventDate: req.eventDate,
    98→            location: req.location,
    99→            price: service?.from ?? req.budgetMax ?? 0,
   100→            summary: req.description,
   101→            message: `Hi ${req.clientName},\n\nI'm excited to send you this custom deal. Please review the details and let me know if you have any questions.`,
   102→            scope: req.description,
   103→          }));
   104→        })
   105→        .catch(() => undefined);
   106→    }
   107→  }, [user, requestId]);
   108→
   109→  const deposit = useMemo(
   110→    () => Math.round((form.price * form.depositPercent) / 100),
   111→    [form.price, form.depositPercent]
   112→  );
   113→  const balance = useMemo(() => form.price - deposit, [form.price, deposit]);
   114→
   115→  function validateStep(current: number): string | null {
   116→    if (current === 0) {
   117→      if (!form.title.trim()) return "Give this deal a clear title.";
   118→      if (!form.clientName.trim()) return "Who is this deal for? Add the client's name.";
   119→      if (!form.clientContact.trim()) return "Add an email or phone so the client can be reached.";
   120→    }
   121→    if (current === 1) {
   122→      if (!form.scope.trim()) return "Describe the scope of work.";
   123→      if (form.deliverables.length === 0) return "Add at least one deliverable.";
   124→    }
   125→    if (current === 2) {
   126→      if (!form.price || form.price <= 0) return "Set a price for this deal.";
   127→      if (!form.dueDate) return "Add a delivery due date.";
   128→    }
   129→    return null;
   130→  }
   131→
   132→  function next() {
   133→    const problem = validateStep(step);
   134→    if (problem) {
   135→      toast.error(problem);
   136→      return;
   137→    }
   138→    setStep((s) => Math.min(s + 1, 3));
   139→  }
   140→
   141→  async function submit(send: boolean) {
   142→    if (!user) return;
   143→    const problem = validateStep(0) ?? validateStep(1) ?? validateStep(2);
   144→    if (problem) {
   145→      toast.error(problem);
   146→      return;
   147→    }
   148→    setBusy(true);
   149→    try {
   150→      const { deal } = await api.createDeal({
   151→        creatorId: user.id,
   152→        requestId: linkedRequest?.id ?? null,
   153→        title: form.title,
   154→        serviceTitle: form.serviceTitle,
   155→        clientName: form.clientName,
   156→        clientContact: form.clientContact,
   157→      });
   158→      await api.updateDeal(deal.id, {
   159→        summary: form.summary,
   160→        eventDate: form.eventDate,
   161→        location: form.location,
   162→        message: form.message,
   163→        scope: form.scope,
   164→        deliverables: form.deliverables,
   165→        price: form.price,
   166→        depositPercent: form.depositPercent,
   167→        startDate: form.startDate,
   168→        dueDate: form.dueDate,
   169→        revisions: form.revisions,
   170→      });
   171→      if (send) {
   172→        const { deal: sent } = await api.dealAction(deal.id, { action: "send" });
   173→        setSentDeal(sent);
   174→        toast.success("Deal sent!");
   175→      } else {
   176→        toast.success("Draft saved", {
   177→          description: "Find it under Projects — you can finish and send it anytime.",
   178→        });
   179→        navigate("/deals");
   180→      }
   181→    } catch (err) {
   182→      toast.error(err instanceof Error ? err.message : "Couldn't save this deal.");
   183→    } finally {
   184→      setBusy(false);
   185→    }
   186→  }
   187→
   188→  /* ---------------- sent confirmation ---------------- */
   189→  if (sentDeal) {
   190→    const shareUrl = `${window.location.origin}/#/c/${sentDeal.shareToken}`;
   191→    const waText = encodeURIComponent(
   192→      `Hi ${sentDeal.client.name}! I've sent you a deal on DEAL for "${sentDeal.title}" (${formatNaira(sentDeal.price)}). Review and accept here: ${shareUrl}`
   193→    );
   194→    return (
   195→      <AppCanvas backHref="/dashboard">
   196→        <div className="flex flex-1 flex-col px-5 pb-10 pt-8">
   197→          <div className="flex justify-center">
   198→            <span className="flex h-20 w-20 items-center justify-center rounded-full bg-accent ring-8 ring-accent/40">
   199→              <Check className="h-9 w-9 text-primary" strokeWidth={3} />
   200→            </span>
   201→          </div>
   202→          <h1 className="mt-5 text-center text-2xl font-extrabold tracking-tight text-foreground">
   203→            Your deal has been <span className="text-primary">sent!</span>
   204→          </h1>
   205→          <p className="mx-auto mt-2 max-w-xs text-center text-[15px] font-medium text-muted-foreground">
   206→            {sentDeal.client.name} will review and accept. You'll be notified once they pay the
   207→            deposit.
   208→          </p>
   209→
   210→          <div className="mt-6 rounded-2xl border border-border bg-card p-4">
   211→            <p className="text-[13px] font-extrabold text-foreground">Client link</p>
   212→            <div className="mt-2 flex items-center gap-2 rounded-xl border border-border bg-muted/60 p-2 pl-3">
   213→              <p className="min-w-0 flex-1 truncate text-[13px] font-bold text-foreground">{shareUrl}</p>
   214→              <Button
   215→                type="button"
   216→                size="sm"
   217→                className="shrink-0 rounded-lg font-bold"
   218→                onClick={() => {
   219→                  navigator.clipboard?.writeText(shareUrl).catch(() => undefined);
   220→                  toast.success("Link copied!");
   221→                }}
   222→              >
   223→                <Copy className="mr-1 h-3.5 w-3.5" /> Copy
   224→              </Button>
   225→            </div>
   226→            <a
   227→              href={`https://wa.me/?text=${waText}`}
   228→              target="_blank"
   229→              rel="noreferrer"
   230→              className="mt-3 flex h-11 w-full items-center justify-center gap-2 rounded-xl border border-primary/30 bg-accent text-sm font-extrabold text-accent-foreground transition-colors hover:bg-primary hover:text-white"
   231→            >
   232→              <svg viewBox="0 0 24 24" className="h-4 w-4 fill-current">
   233→                <path d="M12 2a10 10 0 0 0-8.6 15.1L2 22l5-1.3A10 10 0 1 0 12 2Zm5.5 14.1c-.2.7-1.3 1.3-1.9 1.4-.5.1-1.1.2-3.4-.7-2.9-1.2-4.7-4.1-4.9-4.3-.1-.2-1.1-1.5-1.1-2.9s.7-2 1-2.3c.2-.3.5-.3.7-.3h.5c.2 0 .4 0 .6.4l.9 2.1c.1.2.1.4 0 .6l-.4.6-.5.5c-.1.1-.3.3-.1.6.2.3.8 1.3 1.7 2.1 1.2 1 2.1 1.4 2.4 1.5.3.1.5.1.7-.1l1-1.2c.2-.3.4-.2.7-.1l2 1c.3.1.5.2.6.4 0 .1 0 .7-.5 1.7Z" />
   234→              </svg>
   235→              Share deal on WhatsApp
   236→            </a>
   237→          </div>
   238→
   239→          <Button
   240→            type="button"
   241→            onClick={() => navigate(`/deals/${sentDeal.id}`)}
   242→            className="mt-6 h-13 w-full rounded-xl font-bold"
   243→          >
   244→            View deal <ArrowRight className="ml-1 h-4.5 w-4.5" />
   245→          </Button>
   246→          <Button
   247→            type="button"
   248→            variant="outline"
   249→            onClick={() => navigate("/dashboard")}
   250→            className="mt-3 h-13 w-full rounded-xl font-bold"
   251→          >
   252→            Back to dashboard
   253→          </Button>
   254→        </div>
   255→      </AppCanvas>
   256→    );
   257→  }
   258→
   259→  /* ---------------- wizard ---------------- */
   260→  return (
   261→    <AppCanvas backHref="/dashboard">
   262→      <div className="flex-1 px-5 pb-10 pt-5">
   263→        <div className="flex items-center justify-between">
   264→          <h1 className="text-xl font-extrabold tracking-tight text-foreground">Create DEAL</h1>
   265→          <p className="text-[13px] font-bold text-muted-foreground">
   266→            Step {step + 1} of 4
   267→          </p>
   268→        </div>
   269→
   270→        {/* progress */}
   271→        <ol className="mt-4 flex items-center">
   272→          {WIZARD_STEPS.map((label, i) => (
   273→            <li key={label} className={cn("flex items-center", i < WIZARD_STEPS.length - 1 && "flex-1")}>
   274→              <div className="flex flex-col items-center">
   275→                <span
   276→                  className={cn(
   277→                    "flex h-8 w-8 items-center justify-center rounded-full border-2 text-xs font-extrabold",
   278→                    i < step && "border-primary bg-primary text-white",
   279→                    i === step && "border-primary bg-white text-primary",
   280→                    i > step && "border-border bg-white text-muted-foreground"
   281→                  )}
   282→                >
   283→                  {i < step ? <Check className="h-4 w-4" strokeWidth={3} /> : i + 1}
   284→                </span>
   285→                <span
   286→                  className={cn(
   287→                    "mt-1.5 whitespace-nowrap text-[11px] font-extrabold",
   288→                    i === step ? "text-primary" : "text-muted-foreground"
   289→                  )}
   290→                >
   291→                  {label}
   292→                </span>
   293→              </div>
   294→              {i < WIZARD_STEPS.length - 1 && (
   295→                <span className={cn("mx-2 mb-5 h-0.5 flex-1", i < step ? "bg-primary" : "bg-border")} />
   296→              )}
   297→            </li>
   298→          ))}
   299→        </ol>
   300→
   301→        {/* STEP 1 — deal details */}
   302→        {step === 0 && (
   303→          <div className="mt-6 space-y-4">
   304→            {linkedRequest && (
   305→              <p className="rounded-xl border border-primary/20 bg-accent/60 px-4 py-3 text-[13px] font-bold text-accent-foreground">
   306→                Creating a proposal for {linkedRequest.clientName}'s request ({linkedRequest.ref})
   307→              </p>
   308→            )}
   309→            <div>
   310→              <label className="text-[13px] font-extrabold text-foreground">
   311→                Deal title <span className="text-red-500">*</span>
   312→              </label>
   313→              <Input
   314→                value={form.title}
   315→                onChange={(e) => set("title", e.target.value)}
   316→                placeholder="e.g. Pre-wedding Photography Package"
   317→                className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   318→              />
   319→            </div>
   320→            <div className="grid grid-cols-2 gap-3">
   321→              <div>
   322→                <label className="text-[13px] font-extrabold text-foreground">
   323→                  Client name <span className="text-red-500">*</span>
   324→                </label>
   325→                <Input
   326→                  value={form.clientName}
   327→                  onChange={(e) => set("clientName", e.target.value)}
   328→                  placeholder="e.g. Lola & Tunde"
   329→                  className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   330→                />
   331→              </div>
   332→              <div>
   333→                <label className="text-[13px] font-extrabold text-foreground">
   334→                  Client contact <span className="text-red-500">*</span>
   335→                </label>
   336→                <Input
   337→                  value={form.clientContact}
   338→                  onChange={(e) => set("clientContact", e.target.value)}
   339→                  placeholder="Email or phone"
   340→                  className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   341→                />
   342→              </div>
   343→            </div>
   344→            <div>
   345→              <label className="text-[13px] font-extrabold text-foreground">Short summary</label>
   346→              <Textarea
   347→                value={form.summary}
   348→                onChange={(e) => set("summary", e.target.value.slice(0, 300))}
   349→                placeholder="Briefly describe this deal and what it covers…"
   350→                className="mt-1.5 min-h-20 rounded-xl bg-muted/60 font-semibold"
   351→              />
   352→            </div>
   353→            <div className="grid grid-cols-2 gap-3">
   354→              <div>
   355→                <label className="text-[13px] font-extrabold text-foreground">Event date</label>
   356→                <Input
   357→                  value={form.eventDate}
   358→                  onChange={(e) => set("eventDate", e.target.value)}
   359→                  type="date"
   360→                  className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   361→                />
   362→              </div>
   363→              <div>
   364→                <label className="text-[13px] font-extrabold text-foreground">Location</label>
   365→                <Input
   366→                  value={form.location}
   367→                  onChange={(e) => set("location", e.target.value)}
   368→                  placeholder="e.g. Ikoyi, Lagos"
   369→                  className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   370→                />
   371→              </div>
   372→            </div>
   373→            <div>
   374→              <label className="text-[13px] font-extrabold text-foreground">
   375→                Message to client <span className="font-semibold text-muted-foreground">(optional)</span>
   376→              </label>
   377→              <Textarea
   378→                value={form.message}
   379→                onChange={(e) => set("message", e.target.value.slice(0, 1000))}
   380→                placeholder={`Hi ${form.clientName || "there"}, I'm excited to send you this deal…`}
   381→                className="mt-1.5 min-h-24 rounded-xl bg-muted/60 font-semibold"
   382→              />
   383→            </div>
   384→          </div>
   385→        )}
   386→
   387→        {/* STEP 2 — scope & deliverables */}
   388→        {step === 1 && (
   389→          <div className="mt-6 space-y-4">
   390→            <div>
   391→              <label className="text-[13px] font-extrabold text-foreground">Service</label>
   392→              <div className="mt-2 flex flex-wrap gap-2">
   393→                {services.map((svc) => (
   394→                  <button
   395→                    key={svc.id}
   396→                    type="button"
   397→                    onClick={() => {
   398→                      set("serviceTitle", svc.title);
   399→                      if (!form.price) set("price", svc.from);
   400→                      if (form.deliverables.length === 0 && svc.includes.length) {
   401→                        set("deliverables", svc.includes.slice(0, 4));
   402→                      }
   403→                    }}
   404→                    className={cn(
   405→                      "rounded-full border px-3.5 py-2 text-[13px] font-bold transition-colors",
   406→                      form.serviceTitle === svc.title
   407→                        ? "border-primary bg-accent text-accent-foreground"
   408→                        : "border-border bg-card text-muted-foreground hover:bg-muted"
   409→                    )}
   410→                  >
   411→                    {svc.title}
   412→                  </button>
   413→                ))}
   414→                <button
   415→                  type="button"
   416→                  onClick={() => set("serviceTitle", "Custom service")}
   417→                  className={cn(
   418→                    "rounded-full border px-3.5 py-2 text-[13px] font-bold transition-colors",
   419→                    form.serviceTitle === "Custom service"
   420→                      ? "border-primary bg-accent text-accent-foreground"
   421→                      : "border-dashed border-primary/40 text-primary hover:bg-accent/50"
   422→                  )}
   423→                >
   424→                  + Custom
   425→                </button>
   426→              </div>
   427→            </div>
   428→            <div>
   429→              <label className="text-[13px] font-extrabold text-foreground">
   430→                Scope of work <span className="text-red-500">*</span>
   431→              </label>
   432→              <Textarea
   433→                value={form.scope}
   434→                onChange={(e) => set("scope", e.target.value)}
   435→                placeholder="Describe exactly what you'll do — hours, locations, styles, what's included…"
   436→                className="mt-1.5 min-h-28 rounded-xl bg-muted/60 font-semibold"
   437→              />
   438→            </div>
   439→            <div>
   440→              <label className="text-[13px] font-extrabold text-foreground">
   441→                Deliverables <span className="text-red-500">*</span>
   442→              </label>
   443→              <div className="mt-2 space-y-2">
   444→                {form.deliverables.map((d, i) => (
   445→                  <div
   446→                    key={`${d}-${i}`}
   447→                    className="flex items-center justify-between gap-2 rounded-xl border border-border bg-card px-3.5 py-2.5"
   448→                  >
   449→                    <p className="min-w-0 flex-1 truncate text-[13px] font-bold text-foreground">{d}</p>
   450→                    <button
   451→                      type="button"
   452→                      aria-label={`Remove ${d}`}
   453→                      onClick={() => set("deliverables", form.deliverables.filter((_, j) => j !== i))}
   454→                      className="rounded-md p-1 text-muted-foreground hover:bg-muted"
   455→                    >
   456→                      <X className="h-4 w-4" />
   457→                    </button>
   458→                  </div>
   459→                ))}
   460→                <div className="flex gap-2">
   461→                  <Input
   462→                    value={deliverableDraft}
   463→                    onChange={(e) => setDeliverableDraft(e.target.value)}
   464→                    onKeyDown={(e) => {
   465→                      if (e.key === "Enter" && deliverableDraft.trim()) {
   466→                        e.preventDefault();
   467→                        set("deliverables", [...form.deliverables, deliverableDraft.trim()]);
   468→                        setDeliverableDraft("");
   469→                      }
   470→                    }}
   471→                    placeholder="e.g. 40+ high-resolution edited photos"
   472→                    className="h-11 flex-1 rounded-xl bg-muted/60 font-semibold"
   473→                  />
   474→                  <Button
   475→                    type="button"
   476→                    variant="outline"
   477→                    className="h-11 rounded-xl border-primary/40 px-4 font-bold text-primary"
   478→                    onClick={() => {
   479→                      if (!deliverableDraft.trim()) return;
   480→                      set("deliverables", [...form.deliverables, deliverableDraft.trim()]);
   481→                      setDeliverableDraft("");
   482→                    }}
   483→                  >
   484→                    <Plus className="h-4.5 w-4.5" />
   485→                  </Button>
   486→                </div>
   487→              </div>
   488→            </div>
   489→          </div>
   490→        )}
   491→
   492→        {/* STEP 3 — terms */}
   493→        {step === 2 && (
   494→          <div className="mt-6 space-y-4">
   495→            <div>
   496→              <label className="text-[13px] font-extrabold text-foreground">
   497→                Price (₦) <span className="text-red-500">*</span>
   498→              </label>
   499→              <Input
   500→                value={form.price ? String(form.price) : ""}
   501→                onChange={(e) => set("price", Number(e.target.value.replace(/[^0-9]/g, "")))}
   502→                placeholder="120000"
   503→                inputMode="numeric"
   504→                className="mt-1.5 h-12 rounded-xl bg-muted/60 text-lg font-extrabold"
   505→              />
   506→              <div className="mt-2 flex flex-wrap gap-2">
   507→                {[50000, 80000, 120000, 150000].map((amount) => (
   508→                  <button
   509→                    key={amount}
   510→                    type="button"
   511→                    onClick={() => set("price", amount)}
   512→                    className={cn(
   513→                      "rounded-full border px-3 py-1.5 text-xs font-bold transition-colors",
   514→                      form.price === amount
   515→                        ? "border-primary bg-accent text-accent-foreground"
   516→                        : "border-border bg-card text-muted-foreground"
   517→                    )}
   518→                  >
   519→                    {formatNaira(amount)}
   520→                  </button>
   521→                ))}
   522→              </div>
   523→            </div>
   524→
