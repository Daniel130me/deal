   295→        <motion.div

   589→function formatDayLabel(date: string) {
   590→  const d = new Date(`${date}T00:00:00`);
   591→  if (Number.isNaN(d.getTime())) return date;
   592→  return d.toLocaleDateString("en-GB", { weekday: "short", day: "numeric", month: "short" });
   593→}

   632→function RequestForm({
   633→  creator,
   634→  service,
   635→  onBack,
   636→  onSent,
   637→}: {
   638→  creator: SafeUser;
   639→  service: Service;
   640→  onBack: () => void;
   641→  onSent: (ref: string) => void;
   642→}) {
   643→  const [form, setForm] = useState({
   644→    clientName: "",
   645→    clientContact: "",
   646→    eventDate: "",
   647→    location: "",
   648→    description: "",
   649→    notes: "",
   650→  });
   651→  const [busy, setBusy] = useState(false);
   652→
   653→  async function submit(e: React.FormEvent) {
   654→    e.preventDefault();
   655→    if (busy) return;
   656→    if (!form.clientName.trim() || !form.clientContact.trim() || !form.description.trim()) {
   657→      toast.error("Add your name, contact and a short project description.");
   658→      return;
   659→    }
   660→    setBusy(true);
   661→    try {
   662→      const { request } = await api.sendRequest(creator.handle, {
   663→        serviceId: service.id,
   664→        clientName: form.clientName,
   665→        clientContact: form.clientContact,
   666→        eventDate: form.eventDate,
   667→        location: form.location,
   668→        budgetMin: Math.round(service.from * 0.8),
   669→        budgetMax: Math.round(service.from * 1.3),
   670→        description: form.description,
   671→        notes: form.notes,
   672→      });
   673→      onSent(request.ref);
   674→    } catch (err) {
   675→      toast.error(err instanceof Error ? err.message : "Couldn't send your request.");
   676→    } finally {
   677→      setBusy(false);
   678→    }
   679→  }
   680→
   681→  return (
   682→    <PublicFrame creator={creator} onBook={() => undefined}>
   683→      <form onSubmit={submit} className="flex-1 px-4 pb-12 pt-5 sm:px-6 lg:py-10">
   684→        <div className="mx-auto w-full max-w-md lg:max-w-lg">

   689→          <div className="mt-4 rounded-2xl border border-border bg-card p-4">
   690→            <div className="flex items-start justify-between gap-3">
   691→              <div className="min-w-0">
   692→                <p className="text-[16px] font-extrabold text-foreground">{service.title}</p>
   693→                <p className="mt-0.5 text-[13px] font-medium text-muted-foreground">{service.desc}</p>
   694→              </div>
   695→              <div className="shrink-0 text-right">
   696→                <p className="text-[10px] font-bold uppercase text-muted-foreground">From</p>
   697→                <p className="text-[15px] font-extrabold text-primary">{formatNaira(service.from)}</p>
   698→              </div>
   699→            </div>
   700→          </div>
   701→
   702→          <h1 className="mt-5 text-xl font-extrabold tracking-tight text-foreground lg:text-2xl">
   703→            Tell {creator.name.split(" ")[0]} about your project
   704→          </h1>
   705→          <p className="mt-1 text-[14px] font-medium text-muted-foreground">
   706→            The more details you provide, the better they can understand your needs.
   707→          </p>
   708→
   709→          <div className="mt-5 space-y-4">

   711→              <label htmlFor="rq-name" className="text-[13px] font-extrabold text-foreground">
   712→                Your name
   713→              </label>
   714→              <Input
   715→                id="rq-name"
   716→                value={form.clientName}
   717→                onChange={(e) => setForm({ ...form, clientName: e.target.value })}
   718→                placeholder="e.g. Lola & Tunde"
   719→                className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   720→              />
