     2→

     4→import {

     9→  ImagePlus,

    16→} from "lucide-react";

    20→import { Input } from "@/components/ui/input";
    21→import { Textarea } from "@/components/ui/textarea";
    22→import {
    23→  Select,
    24→  SelectContent,
    25→  SelectItem,
    26→  SelectTrigger,
    27→  SelectValue,
    28→} from "@/components/ui/select";
    29→import {

    34→} from "@/components/ui/dialog";
    35→import { api } from "@/lib/api";

    37→import { CHANNEL_META, CHANNEL_TYPES } from "@/lib/channels";
    38→import { formatNaira, type ChannelType, type CreatorChannel } from "@/lib/types";
    39→import { cn } from "@/lib/utils";
    40→
    41→const ONBOARD_STEPS = ["Profile", "Services", "Preview", "Publish"];
    42→
    43→const CRAFTS = [
    44→  "Photographer",
    45→  "Graphic Designer",
    46→  "Videographer",
    47→  "Writer",
    48→  "Developer",
    49→  "Illustrator",

    52→];
    53→
    54→export default function OnboardingScreen() {
    55→  const { user, setUser, navigate } = useApp();
    56→  const [step, setStep] = useState(0);
    57→  const [busy, setBusy] = useState(false);
    58→
    59→  // profile state
    60→  const [name, setName] = useState(user?.name ?? "");
    61→  const [craft, setCraft] = useState(user?.craft ?? "");
    62→  const [location, setLocation] = useState(user?.location ?? "");
    63→  const [bio, setBio] = useState(user?.bio ?? "");
    64→  const [craftOpen, setCraftOpen] = useState(false);
    65→
    66→  // channels state — "How clients reach you"
    67→  const [channels, setChannels] = useState<CreatorChannel[]>(() => {
    68→    if (user?.channels?.length) return user.channels;
    69→    return user?.whatsapp ? [{ type: "whatsapp", value: user.whatsapp, primary: true }] : [];
    70→  });
    71→  const [channelType, setChannelType] = useState<ChannelType>("whatsapp");
    72→  const [channelValue, setChannelValue] = useState("");
    73→
    74→  // services state (local, seeded from scratch for new creators)
    75→  const [services, setServices] = useState<
    76→    { title: string; desc: string; from: number; duration: string; includes: string[] }[]
    77→  >([]);
    78→  const [addOpen, setAddOpen] = useState(false);
    79→  const [draft, setDraft] = useState({ title: "", desc: "", from: "", duration: "2–3 hrs" });
    80→
    81→  function addChannel() {
    82→    const value = channelValue.trim();
    83→    if (!value) {
    84→      toast.error("Enter a value for the channel (number, username or link).");
    85→      return;
    86→    }
    87→    if (channels.some((c) => c.type === channelType && c.value.toLowerCase() === value.toLowerCase())) {
    88→      toast.error("That channel is already added.");
    89→      return;
    90→    }
    91→    setChannels((prev) => [...prev, { type: channelType, value, primary: prev.length === 0 }]);
    92→    setChannelValue("");
    93→    toast.success(`${CHANNEL_META[channelType].label} added`);
    94→  }
    95→
    96→  function makePrimaryChannel(index: number) {
    97→    setChannels((prev) => prev.map((c, i) => ({ ...c, primary: i === index })));
    98→  }
    99→
   100→  function removeChannel(index: number) {
   101→    setChannels((prev) => {
   102→      const wasPrimary = prev[index]?.primary ?? false;
   103→      return prev
   104→        .filter((_, i) => i !== index)
   105→        .map((c, i) => ({ ...c, primary: wasPrimary ? i === 0 : c.primary }));
   106→    });
   107→  }
   108→
   109→  async function saveProfile() {
   110→    if (!user) return;
   111→    if (!name.trim() || !craft) {
   112→      toast.error("Add your display name and main service first.");
   113→      return;
   114→    }
   115→    setBusy(true);
   116→    try {
   117→      const { user: updated } = await api.updateUser(user.id, {
   118→        name: name.trim(),
   119→        craft,
   120→        location: location.trim(),
   121→        bio: bio.trim(),
   122→        channels,
   123→      });
   124→      setUser(updated);
   125→      setStep(1);
   126→    } catch (err) {
   127→      toast.error(err instanceof Error ? err.message : "Couldn't save profile.");
   128→    } finally {
   129→      setBusy(false);
   130→    }
   131→  }
   132→
   133→  async function saveServicesAndPreview() {
   134→    if (!user) return;
   135→    if (services.length === 0) {
   136→      toast.error("Add at least one service so clients can book you.");
   137→      return;
   138→    }
   139→    setBusy(true);
   140→    try {
   141→      for (const svc of services) {
   142→        await api.addService(user.id, svc);
   143→      }
   144→      setServices([]);
   145→      setStep(2);
   146→    } catch (err) {
   147→      toast.error(err instanceof Error ? err.message : "Couldn't save services.");
   148→    } finally {
   149→      setBusy(false);
   150→    }
   151→  }
   152→
   153→  async function publish() {
   154→    if (!user) return;
   155→    setBusy(true);
   156→    try {
   157→      const { user: updated } = await api.updateUser(user.id, { onboarded: true, channels });
   158→      setUser(updated);
   159→      setStep(3);
   160→    } catch (err) {
   161→      toast.error(err instanceof Error ? err.message : "Couldn't publish your page.");
   162→    } finally {
   163→      setBusy(false);
   164→    }
   165→  }
   166→
   167→  const publicLink = useMemo(
   168→    () => `${typeof window !== "undefined" ? window.location.origin : ""}/#/u/${user?.handle ?? ""}`,
   169→    [user?.handle]
   170→  );
   171→
   172→  return (
   173→    <div className="flex min-h-screen flex-col bg-[#edf3ef]">
   174→      <div className="mx-auto flex w-full max-w-lg flex-1 flex-col bg-background px-5 pb-[max(2rem,env(safe-area-inset-bottom))] pt-5">
   175→        <div className="flex items-center justify-between">
   176→          <Logo />
   177→          <button
   178→            type="button"
   179→            onClick={() => {
   180→              if (user) {
   181→                setUser({ ...user, onboarded: true });
   182→                navigate("/dashboard");
   183→              }
   184→            }}
   185→            className="text-sm font-extrabold text-primary"
   186→          >
   187→            Skip
   188→          </button>
   189→        </div>
   190→
   191→        {/* progress */}
   192→        <ol className="mt-6 flex items-center">
   193→          {ONBOARD_STEPS.map((label, i) => (
   194→            <li key={label} className={cn("flex items-center", i < ONBOARD_STEPS.length - 1 && "flex-1")}>
   195→              <div className="flex flex-col items-center">
   196→                <span
   197→                  className={cn(
   198→                    "flex h-8 w-8 items-center justify-center rounded-full border-2 text-xs font-extrabold",
   199→                    i < step && "border-primary bg-primary text-white",
   200→                    i === step && "border-primary bg-white text-primary",
   201→                    i > step && "border-border bg-white text-muted-foreground"
   202→                  )}
   203→                >
   204→                  {i < step ? <Check className="h-4 w-4" strokeWidth={3} /> : i + 1}
   205→                </span>
   206→                <span
   207→                  className={cn(
   208→                    "mt-1.5 text-[11px] font-extrabold",
   209→                    i === step ? "text-primary" : "text-muted-foreground"
   210→                  )}
   211→                >
   212→                  {label}
   213→                </span>
   214→              </div>
   215→              {i < ONBOARD_STEPS.length - 1 && (
   216→                <span className={cn("mx-2 mb-5 h-0.5 flex-1", i < step ? "bg-primary" : "bg-border")} />
   217→              )}
   218→            </li>
   219→          ))}
   220→        </ol>
   221→
   222→        {/* STEP 1 — profile */}
   223→        {step === 0 && (
   224→          <div className="mt-7 flex-1">
   225→            <h1 className="text-[26px] font-extrabold leading-tight tracking-tight text-foreground">
   226→              Let's set up
   227→              <br />
   228→              your <span className="text-primary">DEAL profile</span>
   229→            </h1>
   230→            <p className="mt-2 text-[15px] font-medium text-muted-foreground">
   231→              This is how clients will see you on your public DEAL page.
   232→            </p>
   233→
   234→            <div className="mt-6 rounded-2xl border border-primary/20 bg-accent/60 p-4">
   235→              <div className="flex items-center gap-4">
   236→                <span className="flex h-14 w-14 items-center justify-center rounded-full bg-white">
   237→                  <ImagePlus className="h-6 w-6 text-primary" />
   238→                </span>
   239→                <div className="min-w-0 flex-1">
   240→                  <p className="text-sm font-extrabold text-foreground">Upload a clear photo of you</p>
   241→                  <p className="text-xs font-medium text-muted-foreground">
   242→                    JPG, PNG or WEBP. Max 5MB.
   243→                  </p>
   244→                </div>
   245→                <Button
   246→                  type="button"
   247→                  size="sm"
   248→                  variant="outline"
   249→                  className="border-primary/40 font-bold text-primary"
   250→                  onClick={() =>
   251→                    toast.info("Photo upload is coming soon", {
   252→                      description: "Your profile works great without it for this demo.",
   253→                    })
   254→                  }
   255→                >
   256→                  Upload
   257→                </Button>
   258→              </div>
   259→            </div>
   260→
   261→            <div className="mt-5 space-y-4">

   264→                <Input
   265→                  value={name}
   266→                  onChange={(e) => setName(e.target.value)}
   267→                  placeholder="e.g. Tobi A."
   268→                  className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   269→                />
   270→              </div>
   271→              <div className="relative">
   272→                <label className="text-[13px] font-extrabold text-foreground">What do you do?</label>
   273→                <button
   274→                  type="button"
   275→                  onClick={() => setCraftOpen((v) => !v)}
   276→                  className="mt-1.5 flex h-12 w-full items-center justify-between rounded-xl border border-input bg-muted/60 px-4 text-left text-[15px] font-semibold text-foreground"
   277→                >
   278→                  <span className={craft ? "" : "text-muted-foreground"}>
   279→                    {craft || "Select your main creative service"}
   280→                  </span>
   281→                  <ChevronDown className="h-4.5 w-4.5 text-muted-foreground" />
   282→                </button>
   283→                {craftOpen && (
   284→                  <ul className="absolute z-20 mt-1 w-full rounded-xl border border-border bg-white p-1 shadow-lg">
   285→                    {CRAFTS.map((c) => (
   286→                      <li key={c}>
   287→                        <button
   288→                          type="button"
   289→                          onClick={() => {
   290→                            setCraft(c);
   291→                            setCraftOpen(false);
   292→                          }}
   293→                          className="w-full rounded-lg px-3 py-2.5 text-left text-sm font-semibold text-foreground hover:bg-accent"
   294→                        >
   295→                          {c}
   296→                        </button>
   297→                      </li>
   298→                    ))}
   299→                  </ul>
   300→                )}
   301→              </div>
   302→              <div>
   303→                <label className="text-[13px] font-extrabold text-foreground">Location</label>
   304→                <Input
   305→                  value={location}
   306→                  onChange={(e) => setLocation(e.target.value)}
   307→                  placeholder="e.g. Akure, Ondo State"
   308→                  className="mt-1.5 h-12 rounded-xl bg-muted/60 font-semibold"
   309→                />
   310→              </div>
   311→              <div>
   312→                <label className="text-[13px] font-extrabold text-foreground">Short bio</label>
   313→                <Textarea
   314→                  value={bio}
   315→                  onChange={(e) => setBio(e.target.value.slice(0, 120))}
   316→                  placeholder="Tell clients a bit about you and your work"
   317→                  className="mt-1.5 min-h-24 rounded-xl bg-muted/60 font-semibold"
   318→                />
   319→                <p className="mt-1 text-right text-xs font-semibold text-muted-foreground">{bio.length}/120</p>
   320→              </div>
   321→
   322→              {/* how clients reach you */}
   323→              <div>
   324→                <label className="text-[13px] font-extrabold text-foreground">How clients reach you</label>
   325→                <p className="mt-0.5 text-xs font-medium text-muted-foreground">
   326→                  Shown on your public page — clients tap to message you directly.
   327→                </p>
   328→                <div className="mt-2 space-y-2">
   329→                  {channels.map((channel, i) => {
   330→                    const meta = CHANNEL_META[channel.type];
   331→                    if (!meta) return null;
   332→                    const Icon = meta.icon;
   333→                    return (
   334→                      <div
   335→                        key={`${channel.type}-${channel.value}-${i}`}
   336→                        className="flex items-center gap-2.5 rounded-xl border border-border bg-card px-3 py-2.5"
   337→                      >
   338→                        <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-accent">
   339→                          <Icon className="h-4 w-4 text-primary" />
   340→                        </span>
   341→                        <div className="min-w-0 flex-1">
   342→                          <p className="text-[13px] font-extrabold text-foreground">{meta.label}</p>
   343→                          <p className="truncate text-xs font-semibold text-muted-foreground">{channel.value}</p>
   344→                        </div>
   345→                        <button
   346→                          type="button"
   347→                          aria-label={channel.primary ? `Primary channel` : `Make ${meta.label} primary`}
   348→                          aria-pressed={channel.primary}
   349→                          onClick={() => makePrimaryChannel(i)}
   350→                          className="rounded-md p-1.5 transition-colors hover:bg-muted"
   351→                        >
   352→                          <Star
   353→                            className={cn(
   354→                              "h-4 w-4",
   355→                              channel.primary ? "fill-amber-400 text-amber-400" : "text-muted-foreground/50"
   356→                            )}
   357→                          />
   358→                        </button>
   359→                        <button
   360→                          type="button"
   361→                          aria-label={`Remove ${meta.label} channel`}
   362→                          onClick={() => removeChannel(i)}
   363→                          className="rounded-md p-1.5 text-muted-foreground transition-colors hover:bg-muted hover:text-red-500"
   364→                        >
   365→                          <X className="h-4 w-4" />
   366→                        </button>
   367→                      </div>
   368→                    );
   369→                  })}
   370→
   371→                  {channels.length === 0 ? (
   372→                    <p className="rounded-xl bg-muted/60 p-3 text-[12px] font-semibold text-muted-foreground">
   373→                      Add at least one way for clients to reach you.
   374→                    </p>
   375→                  ) : null}
   376→
   377→                  <div className="flex gap-2">
   378→                    <Select value={channelType} onValueChange={(v) => setChannelType(v as ChannelType)}>
   379→                      <SelectTrigger
   380→                        aria-label="Channel type"
   381→                        className="h-11 w-[9.5rem] shrink-0 rounded-xl bg-muted/60 font-semibold"
   382→                      >
   383→                        <SelectValue />
   384→                      </SelectTrigger>
   385→                      <SelectContent>
   386→                        {CHANNEL_TYPES.map((t) => (
   387→                          <SelectItem key={t} value={t}>
   388→                            {CHANNEL_META[t].label}
   389→                          </SelectItem>
   390→                        ))}
   391→                      </SelectContent>
   392→                    </Select>
   393→                    <Input
   394→                      value={channelValue}
   395→                      onChange={(e) => setChannelValue(e.target.value)}
   396→                      onKeyDown={(e) => {
   397→                        if (e.key === "Enter") {
   398→                          e.preventDefault();
   399→                          addChannel();
   400→                        }
   401→                      }}
   402→                      placeholder={CHANNEL_META[channelType].placeholder}
   403→                      aria-label="Channel value"
   404→                      className="h-11 min-w-0 flex-1 rounded-xl bg-muted/60 font-semibold"
   405→                    />
   406→                    <Button
   407→                      type="button"
   408→                      variant="outline"
   409→                      aria-label="Add channel"
   410→                      onClick={addChannel}
   411→                      className="h-11 shrink-0 rounded-xl border-primary/40 px-3.5 font-bold text-primary"
   412→                    >
   413→                      <Plus className="h-4.5 w-4.5" />
   414→                    </Button>
   415→                  </div>
   416→                  <p className="text-[11px] font-medium text-muted-foreground">
   417→                    Tap the star to set your primary channel — that&rsquo;s the one shown first.
   418→                  </p>
   419→                </div>
   420→              </div>
   421→            </div>
   422→
   423→            <Button
   424→              type="button"
   425→              onClick={saveProfile}
   426→              disabled={busy}
   427→              className="mt-6 h-13 w-full rounded-xl text-[15px] font-bold"
   428→            >
   429→              {busy ? <Loader2 className="h-4.5 w-4.5 animate-spin" /> : <>Continue <ArrowRight className="ml-1 h-4.5 w-4.5" /></>}
   430→            </Button>
   431→            <p className="mt-3 text-center text-xs font-semibold text-muted-foreground">
   432→              Info you add here can be changed anytime
   433→            </p>
   434→          </div>
   435→        )}
   436→
   437→        {/* STEP 2 — services */}
   438→        {step === 1 && (
   439→          <div className="mt-7 flex-1">
   440→            <h1 className="text-[26px] font-extrabold leading-tight tracking-tight text-foreground">
   441→              What services do
   442→              <br />
   443→              you <span className="text-primary">offer?</span>
   444→            </h1>
   445→            <p className="mt-2 text-[15px] font-medium text-muted-foreground">
   446→              Add the services you want clients to book you for. You can add more or edit later.
   447→            </p>
   448→
   449→            <div className="mt-6 space-y-3">
   450→              {services.map((svc, i) => (
   451→                <div key={i} className="flex items-start justify-between gap-3 rounded-2xl border border-border bg-card p-4">
   452→                  <div className="min-w-0">
   453→                    <p className="text-[15px] font-extrabold text-foreground">{svc.title}</p>
   454→                    <p className="mt-0.5 text-[13px] font-medium leading-relaxed text-muted-foreground">
   455→                      {svc.desc}
   456→                    </p>
   457→                  </div>
   458→                  <div className="shrink-0 text-right">
   459→                    <p className="text-[10px] font-bold uppercase tracking-wide text-muted-foreground">From</p>
   460→                    <p className="text-[15px] font-extrabold text-foreground">{formatNaira(svc.from)}</p>
   461→                  </div>
   462→                </div>

   464→
   465→              {services.length === 0 && (
   466→                <button
   467→                  type="button"
   468→                  onClick={() => setAddOpen(true)}
   469→                  className="w-full rounded-2xl border border-primary/30 bg-accent/50 p-4 text-center transition-colors hover:bg-accent"
   470→                >
   471→                  <p className="flex items-center justify-center gap-2 text-sm font-extrabold text-primary">
   472→                    <Plus className="h-4.5 w-4.5" /> Add your first service

   475→                    e.g. Portrait Session — from ₦50,000
   476→                  </p>
   477→                </button>

   479→
   480→              {services.length > 0 && (
   481→                <button
   482→                  type="button"
   483→                  onClick={() => setAddOpen(true)}
   484→                  className="w-full rounded-2xl border border-dashed border-primary/40 p-4 text-center font-extrabold text-primary transition-colors hover:bg-accent/50"
   485→                >
   486→                  + Add another service
   487→                </button>
   488→              )}
   489→            </div>

   491→            <Button

   493→              onClick={saveServicesAndPreview}

   498→            </Button>
   499→          </div>

   501→
   502→        {/* STEP 3 — preview */}
   503→        {step === 2 && (
