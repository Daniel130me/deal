    92→function heldTotal(deal: Deal) {
    93→  return deal.payments
    94→    .filter((p) => p.status === "held")
    95→    .reduce((sum, p) => sum + p.amount, 0);
    96→}

    98→function lifecycleSteps(deal: Deal): Step[] {
    99→  const delivery: Step["state"] =
   100→    deal.status === "active" || deal.status === "revision" ? "current" : "done";
   101→  const approval: Step["state"] =
   102→    deal.status === "delivered" ? "current" : isApproved(deal) ? "done" : "todo";
   103→  const files: Step["state"] =
   104→    deal.status === "files_released" || deal.status === "completed"
   105→      ? "done"
   106→      : isApproved(deal)
   107→        ? "current"
   108→        : "todo";
   109→  return [
   110→    { label: "Accepted", state: "done" },
   111→    { label: "Payment", sub: "In Payaza escrow", state: "done" },
   112→    { label: "Delivery", sub: deal.status === "revision" ? "Revision" : undefined, state: delivery },
   113→    { label: "Approval", state: approval },
   114→    {
   115→      label: "Final files",
   116→      sub:
   117→        deal.status === "approved" && remainingBalance(deal) > 0 ? "Awaiting balance" : undefined,
   118→      state: files,
   119→    },
   120→  ];
   121→}

   562→        <div className="rounded-2xl border border-primary/20 bg-secondary p-5 text-center lg:p-6">
   563→          <span className="mx-auto flex h-14 w-14 items-center justify-center rounded-full bg-primary ring-8 ring-accent">
   564→            <PartyPopper className="h-7 w-7 text-white" />
   565→          </span>
   566→          <p className="mt-3 text-lg font-extrabold text-foreground">Deal completed!</p>
   567→          <p className="mt-0.5 text-[15px] font-extrabold text-primary">
   568→            {formatNaira(paid)} released to your Payaza payout account
   569→          </p>
   570→          <div className="mt-2 flex justify-center">
   571→            <PayazaMark withText />
   572→          </div>
   573→          <p className="mt-2 text-[13px] font-medium text-muted-foreground">
   574→            Agreed, paid, delivered, approved — DEAL kept the record.
   575→          </p>
   576→        </div>
   577→      );
   578→      break;
   579→
   580→    default:
   581→      banner = (
   582→        <div className="rounded-2xl border border-red-200 bg-red-50 p-4 lg:p-5">
   583→          <p className="text-[15px] font-extrabold text-red-700">
   584→            {deal.status === "declined" ? "Deal declined" : "Dispute raised"}
   585→          </p>
   586→          <p className="mt-0.5 text-[13px] font-medium leading-relaxed text-red-600">
   587→            {deal.status === "declined"
   588→              ? `${deal.client.name} passed on this deal. You can create a new one anytime.`
   589→              : "The DEAL team has been notified and will help both sides resolve it."}
   590→          </p>
   591→          {deal.status === "declined" ? (
   592→            <Button asChild variant="outline" size="sm" className="mt-3 rounded-lg font-bold">
   593→              <HashLink href="#/deals/new">
   594→                <Plus className="mr-1 h-3.5 w-3.5" /> Create a new deal
   595→              </HashLink>
   596→            </Button>
   597→          ) : null}
   598→        </div>
   599→      );
   600→  }
   601→
   602→  return (
   603→    <AppCanvas activeTab="deals" backHref="/deals">
   604→      <AppPage
   605→        title={deal.title}
   606→        chip={<StatusChip status={deal.status} />}
   607→        backHref="#/deals"
   608→        subtitle={
   609→          <>
   610→            <span className="font-mono">{deal.ref}</span> · {deal.client.name}
   611→            {deal.serviceTitle ? ` · ${deal.serviceTitle}` : ""}
   612→          </>
   613→        }
   614→      >
   615→        <motion.div
   616→          initial={{ opacity: 0, y: 12 }}
   617→          animate={{ opacity: 1, y: 0 }}
   618→          transition={{ duration: 0.25, ease: "easeOut" }}
   619→          className="space-y-5 lg:space-y-6"
   620→        >
   621→          {/* hero card */}
   622→          <section className="space-y-5 rounded-2xl border border-border bg-card p-5 lg:p-6">
   623→            <div className="flex flex-wrap items-start justify-between gap-4">
   624→              <MetaRow
   625→                items={[
   626→                  {
   627→                    icon: <CalendarDays className="h-4 w-4" />,
   628→                    text: deal.dueDate ? `Due ${formatDate(deal.dueDate)}` : "Date TBD",
   629→                  },
   630→                  ...(deal.location
   631→                    ? [{ icon: <MapPin className="h-4 w-4" />, text: deal.location }]
   632→                    : []),
   633→                ]}
   634→              />
   635→              <div className="text-right">
   636→                <p className="text-xl font-extrabold text-foreground lg:text-2xl">
   637→                  {formatNaira(deal.price)}
   638→                </p>
   639→                <p className="text-xs font-bold text-muted-foreground">
   640→                  {remaining > 0
   641→                    ? `${formatNaira(paid)} paid · ${formatNaira(remaining)} outstanding`
   642→                    : "Paid in full"}
   643→                </p>
   644→              </div>
   645→            </div>
   646→
   647→            <div>
   648→              <div className="mb-1.5 flex items-center justify-between text-[11px] font-bold text-muted-foreground">
   649→                <span>Payment progress</span>
   650→                <span>{pct}%</span>
   651→              </div>
   652→              <Progress value={pct} className="h-2" />
   653→            </div>
   654→
   655→            {HELD_STATUSES.includes(deal.status) ? <Stepper steps={lifecycleSteps(deal)} /> : null}
   656→
   657→            {banner}
   658→            {actions}
   659→            {deal.status !== "draft" &&
   660→            deal.status !== "sent" &&
   661→            deal.status !== "changes_requested" ? (
   662→              <Button
   663→                asChild
   664→                variant="ghost"
   665→                size="sm"
   666→                className="-ml-2 justify-start self-start text-xs font-bold text-muted-foreground hover:text-foreground"
   667→              >
   668→                <HashLink href={`#/c/${deal.shareToken}`}>
   669→                  <Eye className="mr-1.5 h-3.5 w-3.5" /> Preview what {deal.client.name} sees
   670→                </HashLink>
   671→              </Button>
   672→            ) : null}
   673→          </section>
   674→
   675→          {/* agreement */}
   676→          <div ref={agreementRef} className="scroll-mt-24 space-y-5 lg:space-y-6">
   677→            <AgreementSummary deal={deal} />
   678→            <PaymentSummary deal={deal} />
   679→            {deal.status === "draft" ? (
   680→              <div className="flex flex-col gap-3 rounded-2xl border border-dashed border-primary/40 bg-accent/60 p-4 sm:flex-row sm:items-center sm:justify-between lg:p-5">
   681→                <p className="text-[13px] font-bold text-accent-foreground">
   682→                  Happy with the agreement? Send it to {deal.client.name}.
   683→                </p>
   684→                <Button className="rounded-xl font-bold" disabled={busy} onClick={sendDeal}>
   685→                  {busy ? (
   686→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   687→                  ) : (
   688→                    <Send className="mr-2 h-4 w-4" />
   689→                  )}
   690→                  Send deal
   691→                </Button>
   692→              </div>
   693→            ) : null}
   694→          </div>
   695→
   696→          {/* deliveries */}
   697→          {deal.deliveries.length > 0 ? (
   698→            <SectionCard
   699→              title="Deliveries"
   700→              right={
   701→                <span className="text-xs font-bold text-muted-foreground">
   702→                  {deal.deliveries.length} submission{deal.deliveries.length === 1 ? "" : "s"}
   703→                </span>
   704→              }
   705→            >
   706→              <div className="space-y-5">
   707→                {[...deal.deliveries].reverse().map((delivery, index) => (
   708→                  <div key={delivery.id} className={index > 0 ? "border-t border-border pt-5" : ""}>
   709→                    <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
   710→                      <p className="text-[13px] font-extrabold text-foreground">
   711→                        {index === 0 ? "Latest delivery" : `Delivery ${deal.deliveries.length - index}`}
   712→                      </p>
   713→                      <p className="text-xs font-semibold text-muted-foreground">
   714→                        {formatDateTime(delivery.submittedAt)}
   715→                      </p>
   716→                    </div>
   717→                    {delivery.note ? (
   718→                      <p className="mb-3 rounded-xl bg-muted/60 p-3 text-[13px] font-medium leading-relaxed text-foreground/90">
   719→                        {delivery.note}
   720→                      </p>
   721→                    ) : null}
   722→                    <div className="space-y-2.5">
   723→                      {delivery.files.map((file) => (
   724→                        <FileRow
   725→                          key={file.id}
   726→                          name={file.name}
   727→                          size={file.size}
   728→                          kind={file.kind}
   729→                          onPreview={() =>
   730→                            toast.info("File preview is simulated in this demo", {
   731→                              description:
   732→                                "In production this opens the file in a secure viewer.",
   733→                            })
   734→                          }
   735→                        />
   736→                      ))}
   737→                    </div>
   738→                  </div>
   739→                ))}
   740→              </div>
   741→            </SectionCard>
   742→          ) : null}
   743→
   744→          {/* final files */}
   745→          {deal.finalFiles.length > 0 ? (
   746→            <SectionCard title="Final files">
   747→              <div className="space-y-2.5">
   748→                {deal.finalFiles.map((file) => (
   749→                  <FileRow
   750→                    key={file.id}
   751→                    name={file.name}
   752→                    size={file.size}
   753→                    kind={file.kind}
   754→                    onPreview={() => toast.info("File preview is simulated in this demo")}
   755→                  />
   756→                ))}
   757→              </div>
   758→            </SectionCard>
   759→          ) : null}
   760→
   761→          {/* record */}
   762→          <RecordTimeline events={deal.events} />
   763→        </motion.div>
   764→      </AppPage>
   765→
   766→      {/* deliver dialog */}
   767→      <UploadDialog
   768→        open={deliverOpen}
   769→        onOpenChange={setDeliverOpen}
   770→        title={deal.status === "revision" ? "Submit revised delivery" : "Deliver work for review"}
   771→        hint={`Attach your preview files and add a note for ${deal.client.name}. Final high-resolution files are uploaded after approval.`}
   772→        files={DELIVERY_FILES}
   773→        submitLabel="Submit delivery"
   774→        progress={progress}
   775→        submitting={busy}
   776→        onSubmit={submitDelivery}
   777→      >
   778→        <div>
   779→          <label htmlFor="deliver-note" className="text-[13px] font-extrabold text-foreground">
   780→            Note for {deal.client.name}
   781→          </label>
   782→          <Textarea
   783→            id="deliver-note"
   784→            value={note}
   785→            onChange={(e) => setNote(e.target.value)}
   786→            rows={3}
   787→            placeholder="e.g. First cut of the gallery — have a look at shots 12–24."
   788→            className="mt-1.5 rounded-xl"
   789→          />
   790→        </div>
   791→      </UploadDialog>
   792→
   793→      {/* release final files dialog */}
   794→      <UploadDialog
   795→        open={finalOpen}
   796→        onOpenChange={setFinalOpen}
   797→        title="Upload final files"
   798→        hint="These are the high-resolution files your client will download once you release them."
   799→        files={FINAL_FILES}
   800→        submitLabel="Upload & release"
   801→        progress={progress}
   802→        submitting={busy}
   803→        onSubmit={submitRelease}
   804→      />
   805→    </AppCanvas>
   806→  );
   807→}
