    78→function heldTotal(deal: Deal) {
    79→  return deal.payments
    80→    .filter((p) => p.status === "held")
    81→    .reduce((sum, p) => sum + p.amount, 0);
    82→}

    84→function lifecycleSteps(deal: Deal): Step[] {
    85→  const delivery: Step["state"] =
    86→    deal.status === "active" || deal.status === "revision" ? "current" : "done";
    87→  const approval: Step["state"] =
    88→    deal.status === "delivered" ? "current" : isApproved(deal) ? "done" : "todo";
    89→  const files: Step["state"] =
    90→    deal.status === "files_released" || deal.status === "completed"
    91→      ? "done"
    92→      : isApproved(deal)
    93→        ? "current"
    94→        : "todo";
    95→  return [
    96→    { label: "Accepted", state: "done" },
    97→    { label: "Payment", sub: "In Payaza escrow", state: "done" },
    98→    { label: "Delivery", sub: deal.status === "revision" ? "Revision" : undefined, state: delivery },
    99→    { label: "Approval", state: approval },
   100→    {
   101→      label: "Final files",
   102→      sub:
   103→        deal.status === "approved" && remainingBalance(deal) > 0 ? "Awaiting balance" : undefined,
   104→      state: files,
   105→    },
   106→  ];
   107→}

   556→          <p className="mt-0.5 text-[13px] font-medium leading-relaxed text-red-600">
   557→            {deal.status === "declined"
   558→              ? `${deal.client.name} passed on this deal. You can create a new one anytime.`
   559→              : "The DEAL team has been notified and will help both sides resolve it."}
   560→          </p>
   561→          {deal.status === "declined" ? (
   562→            <Button asChild variant="outline" size="sm" className="mt-3 rounded-lg font-bold">
   563→              <Link href="#/deals/new">
   564→                <Plus className="mr-1 h-3.5 w-3.5" /> Create a new deal
   565→              </Link>
   566→            </Button>
   567→          ) : null}
   568→        </div>
   569→      );
   570→  }
   571→
   572→  return (
   573→    <AppCanvas activeTab="deals" backHref="/deals">
   574→      <AppPage
   575→        title={deal.title}
   576→        chip={<StatusChip status={deal.status} />}
   577→        subtitle={
   578→          <>
   579→            <span className="font-mono">{deal.ref}</span> · {deal.client.name}
   580→            {deal.serviceTitle ? ` · ${deal.serviceTitle}` : ""}
   581→          </>
   582→        }
   583→      >
   584→        <motion.div
   585→          initial={{ opacity: 0, y: 12 }}
   586→          animate={{ opacity: 1, y: 0 }}
   587→          transition={{ duration: 0.25, ease: "easeOut" }}
   588→          className="space-y-5 lg:space-y-6"
   589→        >
   590→          {/* hero card */}
   591→          <section className="space-y-5 rounded-2xl border border-border bg-card p-5 lg:p-6">
   592→            <div className="flex flex-wrap items-start justify-between gap-4">
   593→              <MetaRow
   594→                items={[
   595→                  {
   596→                    icon: <CalendarDays className="h-4 w-4" />,
   597→                    text: deal.dueDate ? `Due ${formatDate(deal.dueDate)}` : "Date TBD",
   598→                  },
   599→                  ...(deal.location
   600→                    ? [{ icon: <MapPin className="h-4 w-4" />, text: deal.location }]
   601→                    : []),
   602→                ]}
   603→              />
   604→              <div className="text-right">
   605→                <p className="text-xl font-extrabold text-foreground lg:text-2xl">
   606→                  {formatNaira(deal.price)}
   607→                </p>
   608→                <p className="text-xs font-bold text-muted-foreground">
   609→                  {remaining > 0
   610→                    ? `${formatNaira(paid)} paid · ${formatNaira(remaining)} outstanding`
   611→                    : "Paid in full"}
   612→                </p>
   613→              </div>
   614→            </div>
   615→
   616→            <div>
   617→              <div className="mb-1.5 flex items-center justify-between text-[11px] font-bold text-muted-foreground">
   618→                <span>Payment progress</span>
   619→                <span>{pct}%</span>
   620→              </div>
   621→              <Progress value={pct} className="h-2" />
   622→            </div>
   623→
   624→            {HELD_STATUSES.includes(deal.status) ? <Stepper steps={lifecycleSteps(deal)} /> : null}
   625→
   626→            {banner}
   627→            {actions}
   628→          </section>
   629→
   630→          {/* agreement */}
   631→          <div ref={agreementRef} className="scroll-mt-24 space-y-5 lg:space-y-6">
   632→            <AgreementSummary deal={deal} />
   633→            <PaymentSummary deal={deal} />
   634→            {deal.status === "draft" ? (
   635→              <div className="flex flex-col gap-3 rounded-2xl border border-dashed border-primary/40 bg-accent/60 p-4 sm:flex-row sm:items-center sm:justify-between lg:p-5">
   636→                <p className="text-[13px] font-bold text-accent-foreground">
   637→                  Happy with the agreement? Send it to {deal.client.name}.
   638→                </p>
   639→                <Button className="rounded-xl font-bold" disabled={busy} onClick={sendDeal}>
   640→                  {busy ? (
   641→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   642→                  ) : (
   643→                    <Send className="mr-2 h-4 w-4" />
   644→                  )}
   645→                  Send deal
   646→                </Button>
   647→              </div>
   648→            ) : null}
   649→          </div>
   650→
   651→          {/* deliveries */}
   652→          {deal.deliveries.length > 0 ? (
   653→            <SectionCard
   654→              title="Deliveries"
   655→              right={
   656→                <span className="text-xs font-bold text-muted-foreground">
   657→                  {deal.deliveries.length} submission{deal.deliveries.length === 1 ? "" : "s"}
   658→                </span>
   659→              }
   660→            >
   661→              <div className="space-y-5">
   662→                {[...deal.deliveries].reverse().map((delivery, index) => (
   663→                  <div key={delivery.id} className={index > 0 ? "border-t border-border pt-5" : ""}>
   664→                    <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
   665→                      <p className="text-[13px] font-extrabold text-foreground">
   666→                        {index === 0 ? "Latest delivery" : `Delivery ${deal.deliveries.length - index}`}
   667→                      </p>
   668→                      <p className="text-xs font-semibold text-muted-foreground">
   669→                        {formatDateTime(delivery.submittedAt)}
   670→                      </p>
   671→                    </div>
   672→                    {delivery.note ? (
   673→                      <p className="mb-3 rounded-xl bg-muted/60 p-3 text-[13px] font-medium leading-relaxed text-foreground/90">
   674→                        {delivery.note}
   675→                      </p>
   676→                    ) : null}
   677→                    <div className="space-y-2.5">
   678→                      {delivery.files.map((file) => (
   679→                        <FileRow
   680→                          key={file.id}
   681→                          name={file.name}
   682→                          size={file.size}
   683→                          kind={file.kind}
   684→                          onPreview={() =>
   685→                            toast.info("File preview is simulated in this demo", {
   686→                              description:
   687→                                "In production this opens the file in a secure viewer.",
   688→                            })
   689→                          }
   690→                        />
   691→                      ))}
   692→                    </div>
   693→                  </div>
   694→                ))}
   695→              </div>
   696→            </SectionCard>
   697→          ) : null}
   698→
   699→          {/* final files */}
   700→          {deal.finalFiles.length > 0 ? (
   701→            <SectionCard title="Final files">
   702→              <div className="space-y-2.5">
   703→                {deal.finalFiles.map((file) => (
   704→                  <FileRow
   705→                    key={file.id}
   706→                    name={file.name}
   707→                    size={file.size}
   708→                    kind={file.kind}
   709→                    onPreview={() => toast.info("File preview is simulated in this demo")}
   710→                  />
   711→                ))}
   712→              </div>
   713→            </SectionCard>
   714→          ) : null}
   715→
   716→          {/* record */}
   717→          <RecordTimeline events={deal.events} />
   718→        </motion.div>
   719→      </AppPage>
   720→
   721→      {/* deliver dialog */}
   722→      <UploadDialog
   723→        open={deliverOpen}
   724→        onOpenChange={setDeliverOpen}
   725→        title={deal.status === "revision" ? "Submit revised delivery" : "Deliver work for review"}
   726→        hint={`Attach your preview files and add a note for ${deal.client.name}. Final high-resolution files are uploaded after approval.`}
   727→        files={DELIVERY_FILES}
   728→        submitLabel="Submit delivery"
   729→        progress={progress}
   730→        submitting={busy}
   731→        onSubmit={submitDelivery}
   732→      >
   733→        <div>
   734→          <label htmlFor="deliver-note" className="text-[13px] font-extrabold text-foreground">
   735→            Note for {deal.client.name}
   736→          </label>
   737→          <Textarea
   738→            id="deliver-note"
   739→            value={note}
   740→            onChange={(e) => setNote(e.target.value)}
   741→            rows={3}
   742→            placeholder="e.g. First cut of the gallery — have a look at shots 12–24."
   743→            className="mt-1.5 rounded-xl"
   744→          />
   745→        </div>
   746→      </UploadDialog>
   747→
   748→      {/* release final files dialog */}
   749→      <UploadDialog
   750→        open={finalOpen}
   751→        onOpenChange={setFinalOpen}
   752→        title="Upload final files"
   753→        hint="These are the high-resolution files your client will download once you release them."
   754→        files={FINAL_FILES}
   755→        submitLabel="Upload & release"
   756→        progress={progress}
   757→        submitting={busy}
   758→        onSubmit={submitRelease}
   759→      />
   760→    </AppCanvas>
   761→  );
   762→}
