    92→function heldTotal(deal: Deal) {
    93→  return deal.payments
    94→    .filter((p) => p.status === "held")
    95→    .reduce((sum, p) => sum + p.amount, 0);
    96→}

    98→function lifecycleSteps(deal: Deal): Step[] {
    99→  const delivery: Step["state"] =
   100→    deal.status === "active" || deal.status === "revision" ? "current" : "done";
   101→  const approval: Step["state"] =
   102→    deal.status === "delivered" ? "current" : isApproved(deal) ? "done" : "todo";
   103→  const files: Step["state"] =
   104→    deal.status === "files_released" || deal.status === "completed"
   105→      ? "done"
   106→      : isApproved(deal)
   107→        ? "current"
   108→        : "todo";
   109→  return [
   110→    { label: "Accepted", state: "done" },
   111→    { label: "Payment", sub: "In Payaza escrow", state: "done" },
   112→    { label: "Delivery", sub: deal.status === "revision" ? "Revision" : undefined, state: delivery },
   113→    { label: "Approval", state: approval },
   114→    {
   115→      label: "Final files",
   116→      sub:
   117→        deal.status === "approved" && remainingBalance(deal) > 0 ? "Awaiting balance" : undefined,
   118→      state: files,
   119→    },
   120→  ];
   121→}

   559→            {formatNaira(paid)} released to your Payaza payout account
   560→          </p>
   561→          <div className="mt-2 flex justify-center">
   562→            <PayazaMark withText />
   563→          </div>
   564→          <p className="mt-2 text-[13px] font-medium text-muted-foreground">
   565→            Agreed, paid, delivered, approved — DEAL kept the record.
   566→          </p>
   567→        </div>
   568→      );
   569→      break;
   570→
   571→    default:
   572→      banner = (
   573→        <div className="rounded-2xl border border-red-200 bg-red-50 p-4 lg:p-5">
   574→          <p className="text-[15px] font-extrabold text-red-700">
   575→            {deal.status === "declined" ? "Deal declined" : "Dispute raised"}
   576→          </p>
   577→          <p className="mt-0.5 text-[13px] font-medium leading-relaxed text-red-600">
   578→            {deal.status === "declined"
   579→              ? `${deal.client.name} passed on this deal. You can create a new one anytime.`
   580→              : "The DEAL team has been notified and will help both sides resolve it."}
   581→          </p>
   582→          {deal.status === "declined" ? (
   583→            <Button asChild variant="outline" size="sm" className="mt-3 rounded-lg font-bold">
   584→              <HashLink href="#/deals/new">
   585→                <Plus className="mr-1 h-3.5 w-3.5" /> Create a new deal
   586→              </HashLink>
   587→            </Button>
   588→          ) : null}
   589→        </div>
   590→      );
   591→  }
   592→
   593→  return (
   594→    <AppCanvas activeTab="deals" backHref="/deals">
   595→      <AppPage
   596→        title={deal.title}
   597→        chip={<StatusChip status={deal.status} />}
   598→        backHref="#/deals"
   599→        subtitle={
   600→          <>
   601→            <span className="font-mono">{deal.ref}</span> · {deal.client.name}
   602→            {deal.serviceTitle ? ` · ${deal.serviceTitle}` : ""}
   603→          </>
   604→        }
   605→      >
   606→        <motion.div
   607→          initial={{ opacity: 0, y: 12 }}
   608→          animate={{ opacity: 1, y: 0 }}
   609→          transition={{ duration: 0.25, ease: "easeOut" }}
   610→          className="space-y-5 lg:space-y-6"
   611→        >
   612→          {/* hero card */}
   613→          <section className="space-y-5 rounded-2xl border border-border bg-card p-5 lg:p-6">
   614→            <div className="flex flex-wrap items-start justify-between gap-4">
   615→              <MetaRow
   616→                items={[
   617→                  {
   618→                    icon: <CalendarDays className="h-4 w-4" />,
   619→                    text: deal.dueDate ? `Due ${formatDate(deal.dueDate)}` : "Date TBD",
   620→                  },
   621→                  ...(deal.location
   622→                    ? [{ icon: <MapPin className="h-4 w-4" />, text: deal.location }]
   623→                    : []),
   624→                ]}
   625→              />
   626→              <div className="text-right">
   627→                <p className="text-xl font-extrabold text-foreground lg:text-2xl">
   628→                  {formatNaira(deal.price)}
   629→                </p>
   630→                <p className="text-xs font-bold text-muted-foreground">
   631→                  {remaining > 0
   632→                    ? `${formatNaira(paid)} paid · ${formatNaira(remaining)} outstanding`
   633→                    : "Paid in full"}
   634→                </p>
   635→              </div>
   636→            </div>
   637→
   638→            <div>
   639→              <div className="mb-1.5 flex items-center justify-between text-[11px] font-bold text-muted-foreground">
   640→                <span>Payment progress</span>
   641→                <span>{pct}%</span>
   642→              </div>
   643→              <Progress value={pct} className="h-2" />
   644→            </div>
   645→
   646→            {HELD_STATUSES.includes(deal.status) ? <Stepper steps={lifecycleSteps(deal)} /> : null}
   647→
   648→            {banner}
   649→            {actions}
   650→          </section>
   651→
   652→          {/* agreement */}
   653→          <div ref={agreementRef} className="scroll-mt-24 space-y-5 lg:space-y-6">
   654→            <AgreementSummary deal={deal} />
   655→            <PaymentSummary deal={deal} />
   656→            {deal.status === "draft" ? (
   657→              <div className="flex flex-col gap-3 rounded-2xl border border-dashed border-primary/40 bg-accent/60 p-4 sm:flex-row sm:items-center sm:justify-between lg:p-5">
   658→                <p className="text-[13px] font-bold text-accent-foreground">
   659→                  Happy with the agreement? Send it to {deal.client.name}.
   660→                </p>
   661→                <Button className="rounded-xl font-bold" disabled={busy} onClick={sendDeal}>
   662→                  {busy ? (
   663→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   664→                  ) : (
   665→                    <Send className="mr-2 h-4 w-4" />
   666→                  )}
   667→                  Send deal
   668→                </Button>
   669→              </div>
   670→            ) : null}
   671→          </div>
   672→
   673→          {/* deliveries */}
   674→          {deal.deliveries.length > 0 ? (
   675→            <SectionCard
   676→              title="Deliveries"
   677→              right={
   678→                <span className="text-xs font-bold text-muted-foreground">
   679→                  {deal.deliveries.length} submission{deal.deliveries.length === 1 ? "" : "s"}
   680→                </span>
   681→              }
   682→            >
   683→              <div className="space-y-5">
   684→                {[...deal.deliveries].reverse().map((delivery, index) => (
   685→                  <div key={delivery.id} className={index > 0 ? "border-t border-border pt-5" : ""}>
   686→                    <div className="mb-3 flex flex-wrap items-center justify-between gap-2">
   687→                      <p className="text-[13px] font-extrabold text-foreground">
   688→                        {index === 0 ? "Latest delivery" : `Delivery ${deal.deliveries.length - index}`}
   689→                      </p>
   690→                      <p className="text-xs font-semibold text-muted-foreground">
   691→                        {formatDateTime(delivery.submittedAt)}
   692→                      </p>
   693→                    </div>
   694→                    {delivery.note ? (
   695→                      <p className="mb-3 rounded-xl bg-muted/60 p-3 text-[13px] font-medium leading-relaxed text-foreground/90">
   696→                        {delivery.note}
   697→                      </p>
   698→                    ) : null}
   699→                    <div className="space-y-2.5">
   700→                      {delivery.files.map((file) => (
   701→                        <FileRow
   702→                          key={file.id}
   703→                          name={file.name}
   704→                          size={file.size}
   705→                          kind={file.kind}
   706→                          onPreview={() =>
   707→                            toast.info("File preview is simulated in this demo", {
   708→                              description:
   709→                                "In production this opens the file in a secure viewer.",
   710→                            })
   711→                          }
   712→                        />
   713→                      ))}
   714→                    </div>
   715→                  </div>
   716→                ))}
   717→              </div>
   718→            </SectionCard>
   719→          ) : null}
   720→
   721→          {/* final files */}
   722→          {deal.finalFiles.length > 0 ? (
   723→            <SectionCard title="Final files">
   724→              <div className="space-y-2.5">
   725→                {deal.finalFiles.map((file) => (
   726→                  <FileRow
   727→                    key={file.id}
   728→                    name={file.name}
   729→                    size={file.size}
   730→                    kind={file.kind}
   731→                    onPreview={() => toast.info("File preview is simulated in this demo")}
   732→                  />
   733→                ))}
   734→              </div>
   735→            </SectionCard>
   736→          ) : null}
   737→
   738→          {/* record */}
   739→          <RecordTimeline events={deal.events} />
   740→        </motion.div>
   741→      </AppPage>
   742→
   743→      {/* deliver dialog */}
   744→      <UploadDialog
   745→        open={deliverOpen}
   746→        onOpenChange={setDeliverOpen}
   747→        title={deal.status === "revision" ? "Submit revised delivery" : "Deliver work for review"}
   748→        hint={`Attach your preview files and add a note for ${deal.client.name}. Final high-resolution files are uploaded after approval.`}
   749→        files={DELIVERY_FILES}
   750→        submitLabel="Submit delivery"
   751→        progress={progress}
   752→        submitting={busy}
   753→        onSubmit={submitDelivery}
   754→      >
   755→        <div>
   756→          <label htmlFor="deliver-note" className="text-[13px] font-extrabold text-foreground">
   757→            Note for {deal.client.name}
   758→          </label>
   759→          <Textarea
   760→            id="deliver-note"
   761→            value={note}
   762→            onChange={(e) => setNote(e.target.value)}
   763→            rows={3}
   764→            placeholder="e.g. First cut of the gallery — have a look at shots 12–24."
   765→            className="mt-1.5 rounded-xl"
   766→          />
   767→        </div>
   768→      </UploadDialog>
   769→
   770→      {/* release final files dialog */}
   771→      <UploadDialog
   772→        open={finalOpen}
   773→        onOpenChange={setFinalOpen}
   774→        title="Upload final files"
   775→        hint="These are the high-resolution files your client will download once you release them."
   776→        files={FINAL_FILES}
   777→        submitLabel="Upload & release"
   778→        progress={progress}
   779→        submitting={busy}
   780→        onSubmit={submitRelease}
   781→      />
   782→    </AppCanvas>
   783→  );
   784→}
