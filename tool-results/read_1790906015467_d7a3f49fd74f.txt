    98→    98→function lifecycleSteps(deal: Deal): Step[] {
    99→    99→  const delivery: Step["state"] =
   100→   100→    deal.status === "active" || deal.status === "revision" ? "current" : "done";
   101→   101→  const approval: Step["state"] =
   102→   102→    deal.status === "delivered" ? "current" : isApproved(deal) ? "done" : "todo";
   103→   103→  const files: Step["state"] =
   104→   104→    deal.status === "files_released" || deal.status === "completed"
   105→   105→      ? "done"
   106→   106→      : isApproved(deal)
   107→   107→        ? "current"
   108→   108→        : "todo";
   109→   109→  return [
   110→   110→    { label: "Accepted", state: "done" },
   111→   111→    { label: "Payment", sub: "In Payaza escrow", state: "done" },
   112→   112→    { label: "Delivery", sub: deal.status === "revision" ? "Revision" : undefined, state: delivery },
   113→   113→    { label: "Approval", state: approval },
   114→   114→    {
   115→   115→      label: "Final files",
   116→   116→      sub:
   117→   117→        deal.status === "approved" && remainingBalance(deal) > 0 ? "Awaiting balance" : undefined,
   118→   118→      state: files,
   119→   119→    },
   120→   120→  ];
   121→   121→}
