    84→    84→function lifecycleSteps(deal: Deal): Step[] {
    85→    85→  const delivery: Step["state"] =
    86→    86→    deal.status === "active" || deal.status === "revision" ? "current" : "done";
    87→    87→  const approval: Step["state"] =
    88→    88→    deal.status === "delivered" ? "current" : isApproved(deal) ? "done" : "todo";
    89→    89→  const files: Step["state"] =
    90→    90→    deal.status === "files_released" || deal.status === "completed"
    91→    91→      ? "done"
    92→    92→      : isApproved(deal)
    93→    93→        ? "current"
    94→    94→        : "todo";
    95→    95→  return [
    96→    96→    { label: "Accepted", state: "done" },
    97→    97→    { label: "Payment", sub: "In Payaza escrow", state: "done" },
    98→    98→    { label: "Delivery", sub: deal.status === "revision" ? "Revision" : undefined, state: delivery },
    99→    99→    { label: "Approval", state: approval },
   100→   100→    {
   101→   101→      label: "Final files",
   102→   102→      sub:
   103→   103→        deal.status === "approved" && remainingBalance(deal) > 0 ? "Awaiting balance" : undefined,
   104→   104→      state: files,
   105→   105→    },
   106→   106→  ];
